def _retry(fn, *args, retries: int = MAX_RETRIES, **kwargs):
    for attempt in range(1, retries + 1):
        try:
            return fn(*args, **kwargs)
        except OperationalError as exc:
            if attempt == retries:
                raise
            wait = attempt * 2
            log.warning(f"   ⚡ DB error (attempt {attempt}/{retries}) — retry in {wait}s: {exc}")
            time.sleep(wait)


def _get_engine():
    engine = create_engine(DB_CONN, pool_pre_ping=True)
    log.info(f"🔌 Connected: {DB_CONN.split('@')[-1]}")
    return engine


def _row_count(engine, table: str) -> int:
    try:
        with engine.connect() as conn:
            return conn.execute(text(f'SELECT COUNT(*) FROM "{table}"')).scalar()
    except Exception:
        return 0


def _read_csv(path: Path) -> pd.DataFrame:
    for enc in ("utf-8-sig", "utf-8", "latin-1"):
        try:
            with open(path, "r", encoding=enc) as f:
                reader   = csv.reader(f)
                header   = next(reader)
                expected = len(header)
                rows = []
                for row in reader:
                    if len(row) == expected:
                        rows.append(row)
                    elif len(row) > expected:
                        rows.append(row[:expected-1] + [" ".join(row[expected-1:])])
                    else:
                        rows.append(row + [""] * (expected - len(row)))
            df = pd.DataFrame(rows, columns=header)
            df.columns = [c.strip().lower() for c in df.columns]
            df = df.rename(columns={"inv_number": "inv_no", "account": "customer_code"})
            log.info(f"📂 {path.name} | {len(df):,} rows | enc={enc}")
            return df
        except UnicodeDecodeError:
            continue
    raise ValueError(f"Cannot read {path}")


def _get_dim_keys(engine, dim_table: str, pk_col: str) -> set:
    try:
        with engine.connect() as conn:
            result = conn.execute(text(f'SELECT "{pk_col}" FROM "{dim_table}"'))
            return {str(r[0]).strip() for r in result if r[0] is not None}
    except Exception as exc:
        log.warning(f"⚠️  Could not fetch {dim_table}.{pk_col}: {exc}")
        return set()

            continue

        cfg_dim_table, cfg_pk, insert_sql = SEED_CONFIG[fact_col]

        fact_vals = set(
            df[fact_col].dropna().astype(str).str.strip().unique()
        ) - {"", "nan", "None", "UNKNOWN", "0", "MISSING"}

        if not fact_vals:
            continue

        existing = _get_dim_keys(engine, cfg_dim_table, cfg_pk)
        missing  = fact_vals - existing

        if not missing:
            log.info(f"   ✅ {cfg_dim_table} — all {len(fact_vals):,} {fact_col} values exist")
            continue

        log.info(f"   🌱 Seeding {len(missing):,} missing {fact_col} values into {cfg_dim_table} ...")
        with engine.begin() as conn:
            for val in sorted(missing):
                conn.execute(text(insert_sql), {"v": val})
        log.info(f"   ✅ Seeded {len(missing):,} rows → {cfg_dim_table}")


def _repair_fk(df: pd.DataFrame, table: str, engine) -> tuple[pd.DataFrame, int]:
    total = 0
    for fact_col, dim_table, dim_pk, fallback in FK_CHECKS.get(table, []):
        if fact_col not in df.columns:
            log.warning(f"⚠️  FK column '{fact_col}' not in {table} — skipping")
            continue
        dim_keys = _get_dim_keys(engine, dim_table, dim_pk)
        if not dim_keys:
            log.warning(f"⚠️  {dim_table} is empty — skipping FK check for {fact_col}")
            continue
        mask = ~df[fact_col].astype(str).str.strip().isin(dim_keys)
        n    = mask.sum()
        if n:
            sample = df.loc[mask, fact_col].unique().tolist()[:10]
            log.warning(f"⚠️  {table}.{fact_col}: {n:,} orphans → replaced with '{fallback}'. Sample: {sample}")
            df.loc[mask, fact_col] = fallback
            total += n
        else:
            log.info(f"✅ FK OK: {table}.{fact_col} → {dim_table}.{dim_pk} ({len(dim_keys):,} keys)")
    return df, total


def _log_audit(engine, table: str, rows: int, orphans: int,
               status: str, elapsed: float = 0.0, notes: str = "") -> None:
    try:
        with engine.begin() as conn:
            conn.execute(text(
                "INSERT INTO etl_audit_log"
                " (table_name, rows_upserted, rows_orphaned, status, elapsed_seconds, notes)"
                " VALUES (:t, :r, :o, :s, :e, :n)"
            ), {"t": table, "r": int(rows), "o": int(orphans),
                "s": status, "e": round(elapsed, 2), "n": notes})
    except Exception as exc:
        log.warning(f"⚠️  Audit log failed: {exc}")


def ensure_schema(engine) -> None:
    for stmt in [s.strip() for s in SCHEMA_SQL.split(";") if s.strip()]:
        try:
            with engine.begin() as conn:
                conn.execute(text(stmt))
        except Exception as e:
            log.warning(f"⚠️ DDL statement failed (ignored): {e}")

    # Seed placeholder rows
    with engine.begin() as conn:
        conn.execute(text("""
            INSERT INTO dim_sales_and_collection_assignment
                (assignment_key, customer_branch_code, month_key, month_start_date, acc_manager, collector)
            VALUES ('UNKNOWN', 'UNKNOWN', 0, '1900-01-01', 'Unknown', 'Unknown')
            ON CONFLICT (assignment_key) DO NOTHING
        """))

    for migration in MIGRATIONS:
        try:
            with engine.begin() as conn:
                conn.execute(text(migration))
        except Exception:
            pass

    log.info("✅ Schema verified / migrated")


def apply_fk_constraints(engine) -> None:
    fk_added = 0
    for fk in FK_STATEMENTS:
        try:
            with engine.begin() as conn:
                conn.execute(text(fk))
            fk_added += 1
            name = fk.split("ADD CONSTRAINT")[1].split("FOREIGN")[0].strip()
            log.info(f"   ✅ FK added: {name}")
        except Exception:
            pass
    if fk_added:
        log.info(f"   FK constraints applied: {fk_added}/{len(FK_STATEMENTS)}")
    else:
        log.info(f"   FK constraints already in place ({len(FK_STATEMENTS)} total)")

def _prep_dim(df: pd.DataFrame, table: str) -> pd.DataFrame:
    renames = {k.lower(): v.lower() for k, v in COL_RENAMES.get(table, {}).items()}
    df = df.rename(columns=renames)
    keep = KEEP_COLS.get(table)
    if keep:
        df = df[[c for c in keep if c in df.columns]]
    return df


def load_dimension(engine, table: str, path: Path) -> int:
    if not path.exists():
        log.warning(f"⚠️  {path.name} not found — skipping {table}")
        return 0

    t0  = time.time()
    df  = _prep_dim(_read_csv(path), table)

    if table == "dim_date":
        int_cols = [
            "date_key","monthnum","year","fiscalmonthnum","curmonthoffset",
            "curquarteroffset","curyearoffset","monthyearnum","weekdaynum",
            "weeksequencenum","curweekoffset","curfiscalyearoffset","dayofyear",
            "currentdayoffset","weekofyear","weekof1stdateofmonth","weekofmonth",
            "totalworkdays","actworkdays","is_friday","is_holiday","is_working_day",
        ]
        for col in int_cols:
            if col in df.columns:
                df[col] = pd.to_numeric(df[col], errors="coerce").fillna(0).astype(int)
        if "date" in df.columns:
            df["date"] = pd.to_datetime(df["date"], errors="coerce").dt.date

    pk_col   = DIM_PK[table]
    all_cols = df.columns.tolist()
    cols_q   = ", ".join(f'"{c}"' for c in all_cols)
    update_q = ", ".join(f'"{c}" = EXCLUDED."{c}"' for c in all_cols if c != pk_col)
    staging  = f"{table}_staging"

    def _do():
        with engine.begin() as conn:
            conn.execute(text(f'DROP TABLE IF EXISTS "{staging}"'))
            df.to_sql(staging, conn, if_exists="replace", index=False,
                      method="multi", chunksize=1000)
            result = conn.execute(text(f"""
                INSERT INTO "{table}" ({cols_q})
                SELECT {cols_q} FROM "{staging}"
                ON CONFLICT ("{pk_col}") DO UPDATE SET {update_q}
            """))
            conn.execute(text(f'DROP TABLE IF EXISTS "{staging}"'))
            return result.rowcount

    affected = _retry(_do)
    elapsed  = round(time.time() - t0, 1)
    total    = _row_count(engine, table)
    log.info(f"✅ {table} → {affected} rows affected | {len(df):,} in CSV | {total:,} total | {elapsed}s")
    _log_audit(engine, table, affected or 0, 0, "SUCCESS", elapsed)
    return total


def load_dim_customer_assignment(engine, path: Path) -> int:
    if not path.exists():
        log.warning(f"⚠️  {path.name} not found — skipping dim_sales_and_collection_assignment")
        return 0

    t0 = time.time()
    df = _read_csv(path)

    required = ['customer_branch_code', 'month_key', 'month_start_date', 'acc_manager', 'collector']
    for col in required:
        if col not in df.columns:
            raise ValueError(f"Missing column {col} in {path.name}")

    df['month_start_date'] = pd.to_datetime(df['month_start_date']).dt.date
    df['month_key'] = pd.to_numeric(df['month_key'], errors='coerce').fillna(0).astype(int)
    df['customer_branch_code'] = df['customer_branch_code'].astype(str).str.strip()
    df['acc_manager'] = df['acc_manager'].astype(str).str.strip().fillna('')
    df['collector'] = df['collector'].astype(str).str.strip().fillna('')

    df['assignment_key'] = df['customer_branch_code'] + '_' + df['month_key'].astype(str)
    df = df[df['customer_branch_code'] != '']
    df = df.drop_duplicates(subset=['assignment_key'], keep='last')

    if df.empty:
        log.info("   No valid rows for dim_sales_and_collection_assignment")
        return 0

    staging = "dim_sales_and_collection_assignment_staging"
    with engine.begin() as conn:
        df.to_sql(staging, conn, if_exists='replace', index=False, method='multi', chunksize=1000)
        conn.execute(text(f"""
            INSERT INTO dim_sales_and_collection_assignment
                (assignment_key, customer_branch_code, month_key, month_start_date, acc_manager, collector)
            SELECT assignment_key, customer_branch_code, month_key, month_start_date, acc_manager, collector
            FROM "{staging}"
            ON CONFLICT (assignment_key) DO UPDATE SET
                customer_branch_code = EXCLUDED.customer_branch_code,
                month_key            = EXCLUDED.month_key,
                month_start_date     = EXCLUDED.month_start_date,
                acc_manager          = EXCLUDED.acc_manager,
                collector            = EXCLUDED.collector
        """))
        conn.execute(text(f'DROP TABLE IF EXISTS "{staging}"'))

    elapsed = round(time.time() - t0, 1)
    log.info(f"✅ dim_sales_and_collection_assignment → {len(df):,} rows upserted | {elapsed}s")
    _log_audit(engine, "dim_sales_and_collection_assignment", len(df), 0, "SUCCESS", elapsed,
               f"upserted {len(df)} rows")
    return len(df)

# ENSURE MONTHLY ASSIGNMENTS BEFORE FACT LOAD
# ─────────────────────────────────────────────────────────────────────────────

def _ensure_monthly_assignments_for_facts(df: pd.DataFrame, table: str, engine) -> None:
    # Handle fact_collected_ageing separately (has month_key directly)
    if table == "fact_collected_ageing":
        if "cust_branch_key" not in df.columns or "month_key" not in df.columns:
            return
        temp = df[["cust_branch_key", "month_key"]].copy()
        temp = temp.dropna()
        if temp.empty:
            return
        temp["cust_branch_key"] = temp["cust_branch_key"].astype(str).str.strip()
        temp["assignment_key"] = temp["cust_branch_key"] + '_' + temp["month_key"].astype(int).astype(str)
        pairs_df = temp[["cust_branch_key", "month_key", "assignment_key"]].drop_duplicates()
    elif table in ("fact_sales_lines", "fact_collection"):
        if "cust_branch_key" not in df.columns or "date_key" not in df.columns:
            return
        temp = df[["cust_branch_key", "date_key"]].copy()
        temp = temp[temp["date_key"].notna() & (temp["date_key"] != 0)]
        if temp.empty:
            return
        temp["month_key"] = (temp["date_key"].astype(int) / 100).astype(int)
        temp["cust_branch_key"] = temp["cust_branch_key"].astype(str).str.strip()
        temp["assignment_key"] = temp["cust_branch_key"] + '_' + temp["month_key"].astype(str)
        pairs_df = temp[["cust_branch_key", "month_key", "assignment_key"]].drop_duplicates()
    else:
        return

    with engine.connect() as conn:
        existing = pd.read_sql(
            "SELECT assignment_key FROM dim_sales_and_collection_assignment",
            conn
        )
    existing_set = set(existing["assignment_key"].astype(str).str.strip())

    missing = pairs_df[~pairs_df["assignment_key"].isin(existing_set)].copy()

    if missing.empty:
        return

    log.info(f"   🧩 Adding {len(missing)} default monthly assignments for {table} ...")

    missing["year"] = missing["month_key"] // 100
    missing["month"] = missing["month_key"] % 100
    missing["month_start_date"] = pd.to_datetime(
        missing["year"].astype(str) + "-" + missing["month"].astype(str) + "-01"
    ).dt.date

    default_rows = pd.DataFrame({
        "assignment_key": missing["assignment_key"],
        "customer_branch_code": missing["cust_branch_key"],
        "month_key": missing["month_key"],
        "month_start_date": missing["month_start_date"],
        "acc_manager": "Not Assigned",
        "collector": "Not Assigned"
    })

    with engine.begin() as conn:
        default_rows.to_sql(
            "_tmp_missing_assign",
            conn,
            if_exists="replace",
            index=False,
            method="multi",
            chunksize=1000
        )
        conn.execute(text("""
            INSERT INTO dim_sales_and_collection_assignment
                (assignment_key, customer_branch_code, month_key, month_start_date, acc_manager, collector)
            SELECT assignment_key, customer_branch_code, month_key, month_start_date, acc_manager, collector
            FROM _tmp_missing_assign
            ON CONFLICT (assignment_key) DO NOTHING
        """))
        conn.execute(text("DROP TABLE IF EXISTS _tmp_missing_assign"))
    log.info("   ✅ Default monthly assignments added.")

# FACT LOADER
# ─────────────────────────────────────────────────────────────────────────────

def _prep_fact(df: pd.DataFrame, table: str) -> pd.DataFrame:
    allowed  = FACT_ALLOWED_COLS[table]
    existing = [c for c in allowed if c in df.columns]
    missing  = set(allowed) - set(existing)
    if missing:
        log.warning(f"⚠️  Columns absent in {table} CSV (will be NULL): {sorted(missing)}")
    df = df[existing].copy()

    for col in FACT_NUMERIC_COLS.get(table, []):
        if col not in df.columns:
            continue
        if col in ("payment_days", "payment_value"):
            df[col] = pd.to_numeric(df[col], errors="coerce").fillna(0).astype(float)
        else:
            df[col] = pd.to_numeric(
                df[col].astype(str).str.replace(",", "", regex=False).str.strip(),
                errors="coerce",
            ).fillna(0)

    # Build assignment_key for fact_sales_lines / fact_collection if missing
    if table in ("fact_sales_lines", "fact_collection"):
        if "date_key" in df.columns and "cust_branch_key" in df.columns:
            month_key_series = (df["date_key"].astype(int) / 100).astype(int)
            df["assignment_key"] = df["cust_branch_key"].astype(str) + '_' + month_key_series.astype(str)

    return df


def load_fact(engine, table: str, path: Path) -> int:
    if not path.exists():
        log.warning(f"⚠️  {path.name} not found — skipping {table}")
        return 0

    _ensure_table_exists(engine, table)

    t0      = time.time()
    df      = _prep_fact(_read_csv(path), table)
    pk_cols = UPSERT_KEYS[table]

    _ensure_monthly_assignments_for_facts(df, table, engine)

    log.info(f"\n🔍 FK integrity check — {table}")
    _seed_missing_dim_values(df, table, engine)
    df, orphans = _repair_fk(df, table, engine)
    log.info(f"📊 After FK repair: {len(df):,} rows | {orphans:,} values replaced")

    missing_pk = [k for k in pk_cols if k not in df.columns]
    if missing_pk:
        raise ValueError(f"❌ Missing PK columns in {table}: {missing_pk}")

    before = len(df)
    df     = df.dropna(subset=pk_cols)
    if len(df) != before:
        log.warning(f"   ⚠️  Dropped {before - len(df):,} rows with NULL PK")

    # Remove month_key from insertion list for fact_sales_lines/collection (generated column)
    all_cols    = [c for c in df.columns if c != "month_key"]
    update_cols = [c for c in all_cols if c not in pk_cols]
    cols_q      = ", ".join(f'"{c}"' for c in all_cols)
    conflict_q  = ", ".join(f'"{c}"' for c in pk_cols)
    update_q    = ", ".join(f'"{c}" = EXCLUDED."{c}"' for c in update_cols)
    staging     = f"{table}_staging"
    upsert_sql  = f"""
        INSERT INTO "{table}" ({cols_q})
        SELECT {cols_q} FROM "{staging}"
        ON CONFLICT ({conflict_q}) DO UPDATE SET {update_q}
    """

    total_chunks   = (len(df) - 1) // CHUNK_SIZE + 1
    total_upserted = 0

    for i, start in enumerate(range(0, len(df), CHUNK_SIZE), 1):
        chunk = df.iloc[start : start + CHUNK_SIZE]
        pct   = round(100 * i / total_chunks)
        log.info(f"   ↳ Chunk {i}/{total_chunks} ({pct}%) — {len(chunk):,} rows")

        def _do_chunk(chunk=chunk):
            with engine.begin() as conn:
                conn.execute(text(f'DROP TABLE IF EXISTS "{staging}"'))
                chunk.to_sql(staging, conn, if_exists="replace",
                             index=False, method="multi", chunksize=1000)
                conn.execute(text(upsert_sql))
                conn.execute(text(f'DROP TABLE IF EXISTS "{staging}"'))

        _retry(_do_chunk)
        total_upserted += len(chunk)

    elapsed = round(time.time() - t0, 1)
    log.info(f"✅ {table} → {total_upserted:,} rows upserted | {orphans:,} orphans repaired | {elapsed}s")
    _log_audit(engine, table, total_upserted, orphans, "SUCCESS", elapsed,
               f"upserted {total_upserted} rows in {total_chunks} chunks")
    return total_upserted
