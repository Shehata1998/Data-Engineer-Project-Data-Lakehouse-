=========================
Extract  →  Transform  →  Load (PostgreSQL)

"""

# ── DAG ───────────────────────────────────────────────────────────────────────
 
with DAG(
    dag_id="erp_reports_full_etl",
    description="ERP ETL — Extract → Transform → Load PostgreSQL",
    default_args=default_args,
    start_date=days_ago(1),
    schedule_interval="0 */3 * * *",
    catchup=False,
    tags=["etl", "salman", "warehouse"],
) as dag:
 
    # ── Extract ───────────────────────────────────────────────────────────────
    t_extract_sales = PythonOperator(
        task_id="extract_sales", python_callable=extract_sales)
 
    t_extract_customer_wise = PythonOperator(
        task_id="extract_customer_wise", python_callable=extract_customer_wise)
 
    t_extract_collection = PythonOperator(
        task_id="extract_collection", python_callable=extract_collection,
        doc_md="Collected Invoice Ageing — extracted for reference")
 
    t_extract_debtors = PythonOperator(
        task_id="extract_debtors", python_callable=extract_debtors,
        doc_md="Debtors Ageing Summary — feeds fact_debtors_ageing")
 
    # ── Transform Facts ───────────────────────────────────────────────────────
    t_transform_sales = PythonOperator(
        task_id="transform_sales", python_callable=transform_sales)
 
    t_transform_collection = PythonOperator(
        task_id="transform_collection", python_callable=transform_collection_airflow)
 
    t_transform_debtors = PythonOperator(
        task_id="transform_debtors",
        python_callable=transform_debtors_ageing_airflow,
        doc_md="Transforms Debtors Ageing CSV → fact_debtors_ageing.csv")
 
    # ── Transform Collected Invoice Ageing (new) ──────────────────────────────
    t_transform_collected_ageing = PythonOperator(
        task_id="transform_collected_ageing",
        python_callable=transform_collected_ageing_airflow,
        doc_md="Transforms Collected Invoice Ageing CSV → fact_collected_ageing.csv"
    )
 
    # ── Transform Monthly Attributes (new) ────────────────────────────────────
    t_transform_customer_monthly = PythonOperator(
        task_id="transform_customer_monthly_attributes",
        python_callable=transform_customer_monthly_airflow,
        doc_md="Builds dim_sales_and_collection_assignment CSV (acc_manager + collector)"
    )
 
    # ── Build Dimensions ──────────────────────────────────────────────────────
    t_download_customer_master = PythonOperator(
        task_id="download_customer_master", python_callable=download_customer_master)
 
    t_build_dim_customer = PythonOperator(
        task_id="build_dim_customer", python_callable=build_dim_customer)
 
    t_build_dim_date = PythonOperator(
        task_id="build_dim_date", python_callable=build_dim_date_airflow)
 
    t_download_item_master = PythonOperator(
        task_id="download_item_master", python_callable=download_item_master)
 
    t_build_dim_item = PythonOperator(
        task_id="build_dim_item", python_callable=build_dim_item)
    t_download_warehouse_master = PythonOperator(
       task_id="download_warehouse_master", python_callable=download_warehouse_master)
    t_download_customer_monthly_attributes = PythonOperator(
       task_id="download_customer_monthly_attributes", python_callable=download_customer_monthly_attributes)
 
    # ── Load ──────────────────────────────────────────────────────────────────
    t_load_dimensions = PythonOperator(
        task_id="load_dimensions", python_callable=load_dimensions,
        doc_md="UPSERT for dim_date, dim_customer, dim_item, dim_warehouse, dim_sales_and_collection_assignment")
 
    t_load_fact_sales = PythonOperator(
        task_id="load_fact_sales", python_callable=load_fact_sales,
        doc_md="Chunked UPSERT fact_sales_lines — PK: (date_key, inv_no, icode, ...)")
 
    t_load_fact_collection = PythonOperator(
        task_id="load_fact_collection", python_callable=load_fact_collection,
        doc_md="Chunked UPSERT fact_collection — PK: (date_key, acode, supplier_code)")
 
    t_load_fact_debtors = PythonOperator(
        task_id="load_fact_debtors", python_callable=load_fact_debtors,
        doc_md="Chunked UPSERT fact_debtors_ageing — PK: (client_code, supplier_code, payment_term)")
 
    t_load_fact_collected_ageing = PythonOperator(
        task_id="load_fact_collected_ageing", python_callable=load_fact_collected_ageing,
        doc_md="Chunked UPSERT fact_collected_ageing — PK: (client_code, supplier_code, inv_no, collection_term)")
 
    # ── Standalone Report (مستقل — بدون علاقات بأي pipeline تاني) ────────────
    t_download_standalone_report = PythonOperator(
        task_id="download_standalone_report",
        python_callable=download_standalone_report,
        doc_md="Download standalone_report.csv from Google Sheets")
 
    t_load_standalone_report = PythonOperator(
        task_id="load_standalone_report",
        python_callable=transform_and_load_standalone,
        doc_md="Full refresh dim_standalone_report — no FK constraints")
 
    # ── Dependencies ──────────────────────────────────────────────────────────
    #
    # ① Sales pipeline
    t_extract_sales >> t_transform_sales
    t_download_warehouse_master >> t_transform_sales
    #
    # ② Collection pipeline (customer_wise needs collection + debtors first)
    t_extract_sales >> t_extract_collection >> t_extract_debtors >> t_extract_customer_wise >> t_transform_collection
    t_download_warehouse_master >> t_extract_collection
    #
    # ③ Debtors pipeline (runs after debtors extract)
    t_extract_debtors >> t_transform_debtors
    #
    # ④ Collected Invoice Ageing pipeline (runs after collection extract)
    t_extract_collection >> t_transform_collected_ageing
    #
    # ⑤ Build dims from transformed sales
    t_transform_sales >> [t_build_dim_customer, t_build_dim_date, t_build_dim_item]
    #
    # ⑥ Collection + Debtors also feed dim_customer
    t_transform_collection >> t_build_dim_customer
    t_transform_debtors    >> t_build_dim_customer
 
    t_download_customer_monthly_attributes >> t_transform_customer_monthly
    #
    # ⑦ Master data enriches dims
    t_download_customer_master >> t_build_dim_customer
    t_download_item_master     >> t_build_dim_item
 
    #
    # ⑧ Transform monthly attributes must happen before load_dimensions (it produces CSV)
    t_transform_customer_monthly >> t_load_dimensions
    #
    # ⑨ Load dims after ALL dims are built (including monthly)
    [t_build_dim_customer, t_build_dim_date, t_build_dim_item] >> t_load_dimensions
    #
    # ⑩ Load facts after dims are loaded (FK safety)
    t_transform_collected_ageing >> t_load_dimensions      >> [t_load_fact_sales, t_load_fact_collection, t_load_fact_debtors, t_load_fact_collected_ageing]
    # ── Standalone Report — مستقل تماماً عن باقي الـ pipeline ────────────────
    # بيشتغل بالتوازي مع أي حاجة تانية — مفيش dependencies
    t_download_standalone_report >> t_load_standalone_report
