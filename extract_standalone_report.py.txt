"""
Download standalone_report.csv from Google Sheets
──────────────────────────────────────────────────
Sheet ID: 1j0EZAk8AZv6bdSYGQkxwoOh1C9ER5TAf6AUAGE9lNoc
- Exports Google Sheet as CSV
- If file already exists → delete it first then re-download
- No API key needed (public shared sheet)
https://docs.google.com/spreadsheets/d/1j0EZAk8AZv6bdSYGQkxwoOh1C9ER5TAf6AUAGE9lNoc/edit?usp=sharing
"""
 
import os
import logging
import requests
from pathlib import Path
 
log = logging.getLogger(__name__)
 
GOOGLE_SHEET_ID    = "1j0EZAk8AZv6bdSYGQkxwoOh1C9ER5TAf6AUAGE9lNoc"
STANDALONE_CSV_PATH = "/opt/airflow/data/standalone_report.csv"
 
EXPORT_URLS = [
    f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/export?format=csv",
    f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/export?format=csv&gid=0",
    f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/pub?output=csv",
]
 
 
def download_standalone_report() -> str:
 
    log.info("🔄 Downloading standalone_report.csv from Google Sheets ...")
 
    Path("/opt/airflow/data").mkdir(parents=True, exist_ok=True)
    os.system("chmod -R 777 /opt/airflow/data")
 
    if Path(STANDALONE_CSV_PATH).exists():
        Path(STANDALONE_CSV_PATH).unlink()
        log.info("🗑️  Old file removed → downloading latest version")
 
    last_error = None
 
    for i, url in enumerate(EXPORT_URLS, 1):
        log.info(f"🔗 Attempt {i}/{len(EXPORT_URLS)}: {url}")
        try:
            response = requests.get(url, stream=True, timeout=60, allow_redirects=True)
 
            if response.status_code != 200:
                last_error = f"Status code: {response.status_code}"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                continue
 
            content_type = response.headers.get("Content-Type", "")
            if "text/html" in content_type:
                last_error = "Google returned HTML (sheet may not be public)"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                continue
 
            with open(STANDALONE_CSV_PATH, "wb") as f:
                for chunk in response.iter_content(chunk_size=32768):
                    if chunk:
                        f.write(chunk)
 
            file_size = Path(STANDALONE_CSV_PATH).stat().st_size
            if file_size < 100:
                last_error = f"File too small ({file_size} bytes)"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                Path(STANDALONE_CSV_PATH).unlink(missing_ok=True)
                continue
 
            with open(STANDALONE_CSV_PATH, "r", encoding="utf-8-sig", errors="replace") as f:
                first_line = f.readline(500)
            if first_line.strip().lower().startswith("<!doctype") or "<html" in first_line.lower():
                last_error = "File content is HTML not CSV"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                Path(STANDALONE_CSV_PATH).unlink(missing_ok=True)
                continue
 
            os.system(f"chmod 777 {STANDALONE_CSV_PATH}")
            log.info(f"✅ standalone_report.csv downloaded | {file_size:,} bytes")
            return STANDALONE_CSV_PATH
 
        except requests.RequestException as exc:
            last_error = str(exc)
            log.warning(f"   ⚠️  {last_error} — trying next URL")
            continue
 
    raise ConnectionError(
        f"❌ Failed after {len(EXPORT_URLS)} attempts. Last error: {last_error}\n"
        f"   Sheet URL: https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/edit"
    )
 
 
if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    download_standalone_report()
