"""
Download customer_monthly_attributes.csv from Google Sheets
────────────────────────────────────────────────────────────
- Fixes permissions on /opt/airflow/data every run
- Downloads the CSV export of the Customer_Master_Report Google Sheet
- If file already exists → delete it first then re-download
- No API key needed (public shared sheet)
 
Related sheets (for reference):
https://docs.google.com/spreadsheets/d/1fgDpoREU4-XUsZNgDUOie33g6ECZgrX--haKu2aJzl0/edit?usp=drive_link  Customer_Master_Report  <- SOURCE
https://docs.google.com/spreadsheets/d/1xW2byCpe46aan79S7YcY6BSKR7QGXzGJlJGKD66vJj0/edit?usp=drive_link  Item_Master_Report
https://docs.google.com/spreadsheets/d/1j0EZAk8AZv6bdSYGQkxwoOh1C9ER5TAf6AUAGE9lNoc/edit?usp=drive_link  Dues_Collection_Fact
https://docs.google.com/spreadsheets/d/1s2juDAPHqjwXu5WiUbnrjWY_Fpw51i1DpB9ZQ0i9EFc/edit?usp=drive_link  warehouse_master
"""
 
import os
import logging
import requests
from pathlib import Path
 
log = logging.getLogger(__name__)
 
# ── Google Sheet source ──────────────────────────────────────────────
GOOGLE_SHEET_ID              = "1fgDpoREU4-XUsZNgDUOie33g6ECZgrX--haKu2aJzl0"  # Customer_Master_Report
# Leave gid unset to default to the first tab. Only set it if the data lives
# on a different tab (copy the real gid= value from that tab's URL — do NOT
# assume it's 0, some sheets don't use 0 for the first tab).
GOOGLE_SHEET_GID             = None
CUSTOMER_MONTHLY_ATTRS_PATH  = "/opt/airflow/data/customer_monthly_attributes.csv"
 
 
def download_customer_monthly_attributes() -> str:
 
    log.info("🔄 Downloading customer_monthly_attributes.csv from Google Sheets ...")
 
    Path("/opt/airflow/data").mkdir(parents=True, exist_ok=True)
    os.system("chmod -R 777 /opt/airflow/data")
    log.info("🔑 Permissions fixed on /opt/airflow/data")
 
    if Path(CUSTOMER_MONTHLY_ATTRS_PATH).exists():
        Path(CUSTOMER_MONTHLY_ATTRS_PATH).unlink()
        log.info("🗑️  Old file removed → downloading latest version")
 
    session = requests.Session()
 
    export_url = f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/export?format=csv"
    if GOOGLE_SHEET_GID:
        export_url += f"&gid={GOOGLE_SHEET_GID}"
 
    log.info(f"🔗 Fetching: {export_url}")
    response = session.get(export_url, stream=True, timeout=60)
 
    if response.status_code != 200:
        raise ConnectionError(
            f"❌ Failed to download customer_monthly_attributes.csv. Status code: {response.status_code}\n"
            f"   تأكد إن الـ Google Sheet مشارك على 'Anyone with the link can view'.\n"
            f"   Sheet ID: {GOOGLE_SHEET_ID} | gid: {GOOGLE_SHEET_GID or 'default (first tab)'}"
        )
 
    content_type = response.headers.get("Content-Type", "")
    if "text/html" in content_type:
        raise PermissionError(
            "❌ Google Sheets returned an HTML page instead of the CSV export.\n"
            "   Make sure sharing is set to 'Anyone with the link can view'.\n"
            f"   Sheet ID: {GOOGLE_SHEET_ID}"
        )
 
    with open(CUSTOMER_MONTHLY_ATTRS_PATH, "wb") as f:
        for chunk in response.iter_content(chunk_size=32768):
            if chunk:
                f.write(chunk)
 
    file_size = Path(CUSTOMER_MONTHLY_ATTRS_PATH).stat().st_size
    if file_size < 50:
        Path(CUSTOMER_MONTHLY_ATTRS_PATH).unlink(missing_ok=True)
        raise ConnectionError(f"❌ Downloaded file too small ({file_size} bytes) — download failed.")
 
    with open(CUSTOMER_MONTHLY_ATTRS_PATH, "r", encoding="utf-8-sig", errors="replace") as f:
        first_line = f.readline(500)
    if first_line.strip().lower().startswith("<!doctype") or "<html" in first_line.lower():
        Path(CUSTOMER_MONTHLY_ATTRS_PATH).unlink(missing_ok=True)
        raise PermissionError(
            "❌ Downloaded file is HTML, not CSV.\n"
            "   Make sure the Google Sheet is shared publicly ('Anyone with the link')."
        )
 
    os.system(f"chmod 777 {CUSTOMER_MONTHLY_ATTRS_PATH}")
    log.info(f"✅ customer_monthly_attributes.csv downloaded successfully")
    log.info(f"📦 File size: {file_size:,} bytes")
    log.info(f"📁 Path: {CUSTOMER_MONTHLY_ATTRS_PATH}")
 
    return CUSTOMER_MONTHLY_ATTRS_PATH
 
 
if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    download_customer_monthly_attributes()
