import logging
import os
import pandas as pd
from pathlib import Path
from sqlalchemy import create_engine, text

log = logging.getLogger(__name__)

INPUT_CSV  = "/opt/airflow/data/standalone_report.csv"
TABLE_NAME = "dim_standalone_report"   # ← غيّر لو عندك اسم تاني

# ── DB connection (نفس إعدادات load_to_postgres) ─────────────────────────────
DB_HOST = os.getenv("POSTGRES_HOST", "postgres")
DB_PORT = os.getenv("POSTGRES_PORT", "5432")
DB_NAME = os.getenv("POSTGRES_DB",   "salman_db")
DB_USER = os.getenv("POSTGRES_USER", "salman")
DB_PASS = os.getenv("POSTGRES_PASSWORD", "salman")


def _get_engine():
    url = f"postgresql+psycopg2://{DB_USER}:{DB_PASS}@{DB_HOST}:{DB_PORT}/{DB_NAME}"
    return create_engine(url, pool_pre_ping=True)


def _read_csv_safe(path: str) -> pd.DataFrame:
    for enc in ("utf-8-sig", "utf-8", "latin-1"):
        try:
            df = pd.read_csv(path, dtype=str, encoding=enc, on_bad_lines="warn")
            df.columns = (
                df.columns.str.strip().str.strip('"')
                .str.lower().str.replace(" ", "_", regex=False)
                .str.replace(r"[^\w]", "_", regex=True)
            )
            return df
        except UnicodeDecodeError:
            continue
    raise ValueError(f"Cannot read {path}")


def _ensure_table(engine, df: pd.DataFrame):
    """بيعمل الجدول لو مش موجود بناءً على columns الـ DataFrame."""
    cols_sql = ",\n    ".join(
        f'"{col}" TEXT' for col in df.columns
    )
    create_sql = f"""
        CREATE TABLE IF NOT EXISTS {TABLE_NAME} (
            _row_id SERIAL PRIMARY KEY,
            {cols_sql}
        );
    """
    with engine.begin() as conn:
        conn.execute(text(create_sql))
    log.info(f"✅ Table {TABLE_NAME} ensured")


def _upsert(engine, df: pd.DataFrame):
    """
    بما إن الجدول ده مفيش PK واضح من الـ Sheet،
    بنعمل TRUNCATE + INSERT كل مرة (full refresh).
    لو عندك PK معروف → غيّر لـ ON CONFLICT DO UPDATE.
    """
    cols = list(df.columns)
    cols_quoted = ", ".join(f'"{c}"' for c in cols)
    placeholders = ", ".join(f":{c}" for c in cols)

    with engine.begin() as conn:
        conn.execute(text(f"TRUNCATE TABLE {TABLE_NAME} RESTART IDENTITY"))
        records = df.where(pd.notna(df), None).to_dict(orient="records")
        if records:
            conn.execute(
                text(f"INSERT INTO {TABLE_NAME} ({cols_quoted}) VALUES ({placeholders})"),
                records,
            )
    log.info(f"✅ {TABLE_NAME} → {len(df):,} rows inserted (full refresh)")


def transform_and_load_standalone(**kwargs) -> str:
    log.info(f"🚀 Loading {TABLE_NAME} ...")

    if not Path(INPUT_CSV).exists():
        raise FileNotFoundError(f"❌ Not found: {INPUT_CSV}")

    # ── قراءة الـ CSV ──────────────────────────────────────────────────
    df = _read_csv_safe(INPUT_CSV)
    log.info(f"📂 Loaded {len(df):,} rows | Columns: {df.columns.tolist()}")

    # ── تنظيف عام ─────────────────────────────────────────────────────
    # شيل الـ rows اللي كلها فاضية
    df.dropna(how="all", inplace=True)
    df.reset_index(drop=True, inplace=True)

    # ── DB ─────────────────────────────────────────────────────────────
    engine = _get_engine()
    _ensure_table(engine, df)
    _upsert(engine, df)

    log.info(f"✅ {TABLE_NAME} loaded | {len(df):,} rows")
    return f"{TABLE_NAME}_loaded"
