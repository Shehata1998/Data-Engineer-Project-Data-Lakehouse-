"""
Download Customer_Master_Report.csv from Google Sheets
──────────────────────────────────────────────────────
- Exports Google Sheet as CSV and downloads it
- If file already exists → delete it first then re-download
- No API key needed (public shared sheet)
 
 
https://docs.google.com/spreadsheets/d/1fgDpoREU4-XUsZNgDUOie33g6ECZgrX--haKu2aJzl0/edit?usp=drive_link Customer_Master_Report
 
 
https://docs.google.com/spreadsheets/d/1xW2byCpe46aan79S7YcY6BSKR7QGXzGJlJGKD66vJj0/edit?usp=drive_link Item_Master_Report
https://docs.google.com/spreadsheets/d/1j0EZAk8AZv6bdSYGQkxwoOh1C9ER5TAf6AUAGE9lNoc/edit?usp=drive_link Dues_Collection_Fact
https://docs.google.com/spreadsheets/d/1s2juDAPHqjwXu5WiUbnrjWY_Fpw51i1DpB9ZQ0i9EFc/edit?usp=drive_link warehouse_master
 
 
"""
 
import os
import logging
import requests
from pathlib import Path
 
log = logging.getLogger(__name__)
 
 
 
GOOGLE_SHEET_ID      = "1fgDpoREU4-XUsZNgDUOie33g6ECZgrX--haKu2aJzl0"
CUSTOMER_MASTER_PATH = "/opt/airflow/data/Customer_Master_Report.csv"
 
# Google Sheets export — بنجرب أكتر من URL لأن gid بيختلف من sheet لتاني
EXPORT_URLS = [
    # الطريقة الأولى: export بدون gid (بياخد الـ sheet الأول تلقائياً)
    f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/export?format=csv",
    # الطريقة التانية: gid=0 (الافتراضي)
    f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/export?format=csv&gid=0",
    # الطريقة التالتة: pub?output=csv
    f"https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/pub?output=csv",
]
 
 
def download_customer_master() -> str:
 
    log.info("🔄 Downloading Customer_Master_Report.csv from Google Sheets ...")
 
    Path("/opt/airflow/data").mkdir(parents=True, exist_ok=True)
    os.system("chmod -R 777 /opt/airflow/data")
    log.info("🔑 Permissions fixed on /opt/airflow/data")
 
    if Path(CUSTOMER_MASTER_PATH).exists():
        Path(CUSTOMER_MASTER_PATH).unlink()
        log.info("🗑️  Old file removed → downloading latest version")
 
    last_error = None
 
    for i, url in enumerate(EXPORT_URLS, 1):
        log.info(f"🔗 Attempt {i}/{len(EXPORT_URLS)}: {url}")
        try:
            response = requests.get(url, stream=True, timeout=60, allow_redirects=True)
 
            if response.status_code != 200:
                last_error = f"Status code: {response.status_code}"
                log.warning(f"   ⚠️  Failed with {last_error} — trying next URL")
                continue
 
            content_type = response.headers.get("Content-Type", "")
            if "text/html" in content_type:
                # Google بيرجّع HTML لما الـ sheet مش public أو الـ URL غلط
                last_error = "Google returned HTML instead of CSV (sheet may not be public)"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                continue
 
            # اكتب الملف
            with open(CUSTOMER_MASTER_PATH, "wb") as f:
                for chunk in response.iter_content(chunk_size=32768):
                    if chunk:
                        f.write(chunk)
 
            # تحقق إن الملف فيه محتوى فعلي (مش HTML مخبّي)
            file_size = Path(CUSTOMER_MASTER_PATH).stat().st_size
            if file_size < 100:
                last_error = f"Downloaded file too small ({file_size} bytes) — likely not CSV"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                Path(CUSTOMER_MASTER_PATH).unlink(missing_ok=True)
                continue
 
            # تحقق إن أول سطر مش HTML
            with open(CUSTOMER_MASTER_PATH, "r", encoding="utf-8-sig", errors="replace") as f:
                first_line = f.readline(500)
            if first_line.strip().lower().startswith("<!doctype") or "<html" in first_line.lower():
                last_error = "File content is HTML, not CSV"
                log.warning(f"   ⚠️  {last_error} — trying next URL")
                Path(CUSTOMER_MASTER_PATH).unlink(missing_ok=True)
                continue
 
            os.system(f"chmod 777 {CUSTOMER_MASTER_PATH}")
            log.info(f"✅ Customer_Master_Report.csv downloaded successfully")
            log.info(f"📦 File size: {file_size:,} bytes")
            log.info(f"📁 Path: {CUSTOMER_MASTER_PATH}")
            return CUSTOMER_MASTER_PATH
 
        except requests.RequestException as exc:
            last_error = str(exc)
            log.warning(f"   ⚠️  Request error: {last_error} — trying next URL")
            continue
 
    # كل المحاولات فشلت
    raise ConnectionError(
        f"❌ Failed to download Customer_Master_Report.csv after {len(EXPORT_URLS)} attempts.\n"
        f"   Last error: {last_error}\n"
        f"   تأكد إن الـ Google Sheet sharing مضبوط على 'Anyone with the link can view'.\n"
        f"   Sheet URL: https://docs.google.com/spreadsheets/d/{GOOGLE_SHEET_ID}/edit"
    )
 
 
if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    download_customer_master()
